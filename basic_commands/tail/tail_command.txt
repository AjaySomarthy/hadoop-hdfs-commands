-- Display the last 1kb of a file
hdfs dfs -tail /demo/file1.txt
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- Displays last 1 line of the file
hadoop fs -cat /demo/file1.txt | tail -n -1

-- Displays last 2 lines of the file
hadoop fs -cat /demo/file1.txt | tail -n -2
------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
-- Displays the last 1 file or subdirectory of a directory
hadoop fs -ls /demo/ | tail -1

-- Displays the last 2 files or subdirectories of a directory
hadoop fs -ls /demo/ | tail -2
